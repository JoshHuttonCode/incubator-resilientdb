26TEST(CompositeKeyCodecTest, RoundTrip) {
27 std::string encoded =
28 EncodeCompositeKey("byOwner", {"alice", "active"}, "user:1");
29 EXPECT_FALSE(encoded.empty());
30
31 std::string idx, pk;
32 std::vector<std::string> attrs;
33 ASSERT_TRUE(DecodeCompositeKey(encoded, &idx, &attrs, &pk));
34 EXPECT_EQ(idx, "byOwner");
35 EXPECT_EQ(attrs.size(), 2u);
36 EXPECT_EQ(attrs[0], "alice");
37 EXPECT_EQ(attrs[1], "active");
38 EXPECT_EQ(pk, "user:1");
39}

41TEST(CompositeKeyCodecTest, RejectDelimInInput) {
42 // C-string literals truncate at '\0'; build the strings explicitly.
43 std::string bad_index = std::string("in") + kCompositeKeyDelim + "dex";
44 EXPECT_TRUE(EncodeCompositeKey(bad_index, {"attr"}, "pk").empty());
45
46 std::string bad_attr = std::string("at") + kCompositeKeyDelim + "tr";
47 EXPECT_TRUE(EncodeCompositeKey("idx", {bad_attr}, "pk").empty());
48
49 std::string bad_pk = std::string("p") + kCompositeKeyDelim + "k";
50 EXPECT_TRUE(EncodeCompositeKey("idx", {"attr"}, bad_pk).empty());
51}

53TEST(CompositeKeyCodecTest, PrefixIsStrictBytePrefix) {
54 std::string full = EncodeCompositeKey("byOwner", {"alice"}, "user:1");
55 std::string prefix = EncodeCompositeKeyPrefix("byOwner", {"alice"});
56
57 EXPECT_FALSE(full.empty());
58 EXPECT_FALSE(prefix.empty());
59 EXPECT_EQ(full.find(prefix), 0u);
60 EXPECT_LT(prefix.size(), full.size());
61}
